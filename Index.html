function postAction(action, dataPayload) { if(tg.HapticFeedback) tg.HapticFeedback.impactOccurred('medium'); switchScreen('loading-screen'); const body = Object.assign({ initData: tg.initData, action: action }, dataPayload); // FIX: Added redirect: 'follow' to prevent CORS and HTML token errors fetch(GAS_WEBHOOK_URL, { method: 'POST', redirect: 'follow', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify(body) }) .then(res => res.json()) .then(p => { if (p.success) fetchCharacterData(); else { tg.showAlert(p.message); fetchCharacterData(); } }) .catch(err => { document.getElementById('debug-log').style.display="block"; document.getElementById('debug-log').innerText = `Action failed: ${err.message}`; switchScreen('sheet-screen'); }); }
